  LED_circle 灯臂中轴进度条显示函数 (核心修改)

void LED_circle(uint16_t leaf)
{
     如果该扇叶状态是 all_finish (通关)，则灯臂全亮
    if(LED[leaf].mode == all_finish)
    {
        for(int i = CIRCLE6_START + 1; i = MY_WS2812_MAX_NUM; i++)
        {
            my_ws2812_1_set(leaf, i, 222, 0, 0);  全红
        }
        return;
    }

     否则，根据 game_stage 显示进度
     假设灯臂区域是从 CIRCLE6_START + 1 到 MY_WS2812_MAX_NUM
     你需要确认头文件中这些宏的定义，这里假设它们定义了灯臂的范围
    
    int arm_start_index = CIRCLE6_START + 1;
    int arm_end_index = MY_WS2812_MAX_NUM;
    int total_arm_leds = arm_end_index - arm_start_index + 1;

     计算应该亮多少颗灯 (05 到 55)
    int leds_to_light = 0;
    if (game_stage  0)
    {
         比例计算： (当前阶段  5.0)  总灯数
        leds_to_light = (int)((float)game_stage  5.0f  (float)total_arm_leds);
    }

     循环设置灯珠
    for (int i = 0; i  total_arm_leds; i++)
    {
        int current_led_index = arm_start_index + i;

        if (i  leds_to_light)
        {
             进度范围内的灯 - 亮起 (红色)
            my_ws2812_1_set(leaf, current_led_index, 222, 0, 0);
        }
        else
        {
             超出进度的灯 - 熄灭
            my_ws2812_1_set(leaf, current_led_index, 0, 0, 0);
        }
    }
}
